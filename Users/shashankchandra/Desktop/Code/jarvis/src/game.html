let score = 0; let attempts = 5; function startGame() { const gameBoard = document.getElementById('game-board'); gameBoard.innerHTML = ''; // Generate a random number for the player to guess let randomNumber = Math.floor(Math.random() * 100); let computerNumber = randomNumber; function guess(number) { if (number < computerNumber) { attempts--; result.textContent = "Too low!"; } else if (number > computerNumber) { attempts--; result.textContent = "Too high!"; } if (attempts === 0 || number === computerNumber) { gameOver(); } else { document.getElementById('guess-input').focus(); } } const button = document.createElement('button'); button.textContent = `Guess a number!`; button.onclick = function() { guess(parseInt(document.getElementById('guess-input').value)); }; gameBoard.appendChild(button); // Show the scoreboard document.getElementById('scoreboard').style.display = 'block'; } function gameOver() { result.textContent = "Game over!"; document.getElementById('game-board').style.display = 'none'; let scoreElement = document.getElementById('score'); scoreElement.textContent = `Final Score: ${score}`; const scoreboard = document.getElementById('scoreboard'); scoreboard.style.display = 'block'; } function updateScoreboard() { const scoreElement = document.getElementById('score'); scoreElement.textContent = `Score: ${score}`; }
